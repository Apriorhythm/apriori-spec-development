   277	function containsReal(root, target) {
   278	  let realRoot;
   279	  try { realRoot = fs.realpathSync(root); } catch { return false; }
   280	  let cur = path.resolve(target);
   281	  const rest = [];
   282	  while (!fs.existsSync(cur)) {
   283	    const parent = path.dirname(cur);
   284	    if (parent === cur) return false;
   285	    rest.unshift(path.basename(cur));
   286	    cur = parent;
   287	  }
   288	  let realCur;
   289	  try { realCur = fs.realpathSync(cur); } catch { return false; }
   290	  const full = rest.length ? path.join(realCur, ...rest) : realCur;
   291	  return full !== realRoot && (full + path.sep).startsWith(realRoot + path.sep);
   292	}

   294	function mdFilesUnder(dir) {
   295	  const out = [];
   296	  const walk = (d) => {
   297	    for (const e of fs.readdirSync(d, { withFileTypes: true })) {
   298	      const p = path.join(d, e.name);
   299	      // Dirents reflect lstat: a symlink is neither file nor directory. Follow it explicitly —
   300	      // silently skipping a symlinked dir would hide deltas (fail-open); containment judges the target.
   301	      let isDir = e.isDirectory();
   302	      if (e.isSymbolicLink()) {
   303	        try { isDir = fs.statSync(p).isDirectory(); } catch { isDir = false; /* broken link: fall through, read fails loudly */ }
   304	      }
   305	      if (isDir) walk(p);
   306	      else if (e.name.endsWith('.md')) out.push(p);
   307	    }
   308	  };
   309	  if (fs.existsSync(dir)) walk(dir);
   310	  return out.sort();

   313	// Discover a change's delta files. → { files: [{abs, suffix}], errors: [] } — errors are
   314	// VALIDATION class (exit 2 on both surfaces): bad name, missing dir, escape, zero files.
   315	function discoverDeltas(changesDirAbs, name) {
   316	  const errors = [];

   318	  if (!vName.ok) return { files: [], errors: [`invalid change name '${name}' (${vName.kind}) — bare kebab-case, not date-prefixed, not a reserved name`] };
   319	  const changeDir = path.join(changesDirAbs, name);
   320	  if (!fs.existsSync(changeDir)) return { files: [], errors: [`change '${name}' not found under ${changesDirAbs}`] };
   321	  if (!containsReal(changesDirAbs, changeDir)) return { files: [], errors: [`change path escapes ${changesDirAbs}: ${changeDir}`] };
   322	  const specsDir = path.join(changeDir, 'specs');
   323	  // a symlinked specs/ pointing outside would smuggle its target in as the containment ROOT —
   324	  // require specs/ itself to really live inside the change dir before trusting it as a root
   325	  if (fs.existsSync(specsDir) && !containsReal(changeDir, specsDir))
   326	    return { files: [], errors: [`specs dir escapes the change dir (symlink?): ${specsDir}`] };
   327	  const all = mdFilesUnder(specsDir);
   328	  if (!all.length) return { files: [], errors: [`no delta spec files (*.md) found under ${specsDir}`] };
   329	  const files = [];

   331	    if (!containsReal(specsDir, abs)) { errors.push(`delta path escapes the change root: ${abs}`); continue; }

   337	// Build the projection: apply every delta to its mapped store file IN MEMORY.
   338	// → { texts: Map<suffix, projectedText>, modules, perModule, conflicts, casMismatches,
   339	//     hygiene, validation } — texts covers ALL store files (pass-through + projected).
   340	function buildProjection(storeRoot, deltaFiles, change) {
   341	  const texts = new Map(), perModule = new Map(), modules = [];
   342	  const conflicts = [], casMismatches = [], hygiene = [], validation = [];
   343	  const unstampedMutations = [], notes = [], repaired = [];
   344	  for (const abs of mdFilesUnder(storeRoot))
   345	    texts.set(path.relative(storeRoot, abs).split(path.sep).join('/'), fs.readFileSync(abs, 'utf8'));
   346	  for (const { abs, suffix } of deltaFiles) {
   347	    modules.push(suffix);
   348	    const target = path.join(storeRoot, ...suffix.split('/'));
   349	    if (!containsReal(storeRoot, target)) { validation.push(`store target escapes ${storeRoot}: ${target}`); continue; }
   350	    const raw = fs.readFileSync(abs, 'utf8');
   351	    if (!raw.trim()) { hygiene.push(`empty delta file: ${abs}`); continue; }
   352	    const { delta, stamp, problems } = parseDeltaStrict(raw);
   353	    if (problems.length) { hygiene.push(...problems.map((p) => `${abs}: ${p}`)); continue; }
   354	    if (deltaOpCount(delta) === 0) { hygiene.push(`parsed 0 delta operations from ${abs} — it has content but no recognized operations`); continue; }
   355	    if (!stamp && mutationOpCount(delta) > 0) unstampedMutations.push(suffix);   // WARN grade (callers print)
   356	    const storeText = texts.get(suffix) || '';
   357	    if (stamp) {
   358	      const actual = texts.has(suffix) ? fingerprint(storeText) : 'new';
   359	      if (stamp !== actual) {
   360	        // rerun repair: a mismatch whose every op is already applied is the resumed-rerun
   361	        // signature — divergence with ANY real pending op still fails exactly as before
   362	        const probe = merge(storeText, delta, change);
   363	        const pending = probe.merged.length + probe.modified.length + probe.deprecated.length + probe.renamed.length;
   364	        if (pending === 0 && probe.conflicts.length === 0) {
   365	          notes.push(`${suffix}: stamp mismatch but the delta is already fully applied — rerun accepted`);
   366	          perModule.set(suffix, probe);
   367	          repaired.push(suffix);
   368	          continue;                                          // contributes no writes
   369	        }
   370	        casMismatches.push(`base mismatch for ${target}: delta expects ${stamp}, store is ${actual}`); continue;
   371	      }
   372	    }
   373	    const r = merge(storeText, delta, change);
   374	    perModule.set(suffix, r);
   375	    if (r.conflicts.length) { conflicts.push(...r.conflicts.map((c) => `${suffix}: ${c}`)); continue; }
   376	    texts.set(suffix, renderStore(storeText, r.store));
   377	  }
   378	  modules.sort();
   379	  return { texts, modules, perModule, conflicts, casMismatches, hygiene, validation, unstampedMutations, notes, repaired };
   380	}

   411	// High-level archive: the whole change, four phases (preflight → stage → commit → move).
   412	// → { code, out: [], err: [] } — pure return; cli() prints. ops is the fault-injection seam.
   413	function archiveChange(o) {
   414	  const out = [], err = [];
   415	  const ops = o.ops || { writeFileSync: fs.writeFileSync.bind(fs), renameSync: fs.renameSync.bind(fs), rmSync: fs.rmSync.bind(fs) };
   416	  const cwd = o.cwd || process.cwd();
   417	  const changesDir = path.resolve(cwd, o.changesDir || path.join('apriori', 'changes'));
   418	  const storeRoot = path.resolve(cwd, 'apriori', 'specs');
   419	
   420	  // --- phase 1: preflight (no writes) ---
   421	  const d = discoverDeltas(changesDir, o.change);
   422	  if (d.errors.length) { err.push(...d.errors.map((e) => `archive: ${e}`)); return { code: 2, out, err }; }
   423	  const p = buildProjection(storeRoot, d.files, o.change);
   424	  if (p.validation.length) { err.push(...p.validation.map((e) => `archive: ${e}`)); return { code: 2, out, err }; }
   425	  // per-module report for whatever merged cleanly (dry-run info comes first, then failures)
   426	  for (const suffix of p.modules) {
   427	    const r = p.perModule.get(suffix);
   428	    if (!r || r.conflicts.length) continue;
   429	    out.push(`--- ${suffix}`);
   430	    for (const [label, key] of CATEGORY_LABELS) if (r[key].length) out.push(`${label}: ${r[key].join(', ')}`);
   431	  }
   432	  const casDenials = [];
   433	  if (p.unstampedMutations.length) {
   434	    const w = casWaiver(cwd, o.noCas);
   435	    if (w.problem) {
   436	      casDenials.push(`${w.problem} — a broken config never equals a waiver (denied)`);
   437	    } else if (!w.waiver) {
   438	      for (const sfx of p.unstampedMutations)
   439	        casDenials.push(`${unstampedWarning(sfx)} — denied by default; waive with --no-cas or a | cas | optional | config row`);
   440	    } else {
   441	      for (const sfx of p.unstampedMutations) out.push(`warning: ${unstampedWarning(sfx)}`);
   442	      out.push(`note: cas waived by ${w.waiver}`);
   443	    }
   444	  }
   445	  for (const n of p.notes) out.push(`note: ${n}`);
   446	  // CAS denial joins the preflight failure set — every diagnosis stays visible in one report
   447	  const preflightFailures = [...casDenials, ...p.hygiene, ...p.casMismatches, ...p.conflicts];
   448	  if (preflightFailures.length) {
   449	    for (const f of preflightFailures) err.push(`archive: ${f}`);
   450	    out.push('\nRESULT: FAILED PREFLIGHT — nothing written');
   451	    return { code: 1, out, err };
   452	  }
   453	  const repaired = new Set(p.repaired);                     // already applied — no writes for these
   454	  const jobs = d.files.filter(({ suffix }) => !repaired.has(suffix)).map(({ suffix }) => {
   455	    const store = path.join(storeRoot, ...suffix.split('/'));
   456	    return { suffix, store, tmp: store + '.tmp-archive', text: p.texts.get(suffix) };
   457	  }).sort((a, b) => (a.store < b.store ? -1 : 1));
   458	  if (o.write) {
   459	    for (const j of jobs) if (fs.existsSync(j.tmp)) {
   460	      err.push(`archive: temp file already exists: ${j.tmp} — another run in flight or a manual-recovery artifact; not touching it`);
   461	      out.push('\nRESULT: FAILED PREFLIGHT — nothing written');
   462	      return { code: 1, out, err };
   463	    }
   464	    if (o.changesDirExplicit) {
   465	      // destination containment is checked before any write, not at move time
   466	      const probe = path.join(changesDir, 'archive', 'probe');
   467	      if (!containsReal(changesDir, probe)) {
   468	        err.push(`archive: archive destination escapes ${changesDir} (symlinked archive/?)`);
   469	        return { code: 2, out, err };
   470	      }
   471	    }
   472	  }
   473	  if (!o.write) {
   474	    out.push(`\nRESULT: MERGED (dry-run; ${jobs.length} module(s)) — pass --write to apply`);
   475	    return { code: 0, out, err };
   476	  }
   477	
   478	  // --- phase 2: stage (this run's temps only — preflight proved none pre-existed) ---
   479	  const staged = [];
   480	  try {
   481	    for (const j of jobs) {
   482	      fs.mkdirSync(path.dirname(j.store), { recursive: true });
   483	      ops.writeFileSync(j.tmp, j.text);
   484	      staged.push(j.tmp);
   485	    }
   486	  } catch (e) {
   487	    for (const t of staged) { try { ops.rmSync(t, { force: true }); } catch { /* best effort */ } }
   488	    err.push(`archive: staging failed (${e.message}) — stores untouched, this run's temp files removed`);
   489	    return { code: 1, out, err };
   490	  }
   491	
   492	  // --- phase 3: commit (per-file atomic renames, sorted order; no rollback on mid-commit failure) ---
   493	  const committed = [];
   494	  for (const j of jobs) {
   495	    try { ops.renameSync(j.tmp, j.store); committed.push(j); }
   496	    catch (e) {
   497	      err.push(`archive: COMMIT FAILED at ${j.suffix} (${e.message})`);
   498	      err.push(`  committed:     ${committed.map((c) => c.suffix).join(', ') || '(none)'}`);
   499	      const remaining = jobs.filter((x) => !committed.includes(x));
   500	      err.push(`  NOT committed: ${remaining.map((x) => x.suffix).join(', ')}`);
   501	      err.push(`  temp files remaining for manual completion: ${remaining.map((x) => x.tmp).join(', ')}`);
   502	      return { code: 1, out, err };
   503	    }
   504	  }
   505	
   506	  // --- phase 4: move (only with explicit --changes-dir) ---
   507	  let moved = null;
   508	  if (o.changesDirExplicit) {
   509	    try { moved = archiveChangeDir(changesDir, o.change, new Date(), ops); }
   510	    catch (e) { err.push(`archive: stores committed but the change-dir move failed: ${e.message} — rerun to complete`); return { code: 1, out, err }; }
   511	    if (moved === null) { err.push('archive: stores committed but the change dir vanished before the move'); return { code: 1, out, err }; }
   512	  }
   513	  out.push(`\nRESULT: MERGED — ${jobs.length} module store(s) rewritten${moved ? `; change archived → ${moved}` : ''}`);
   514	  return { code: 0, out, err };
   515	}
