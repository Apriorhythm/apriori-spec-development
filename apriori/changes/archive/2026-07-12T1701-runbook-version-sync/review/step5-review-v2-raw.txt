// ---- CK-06: scaffolded-runbook freshness (warn-only, never a failure) ---------
// ---- CK-11: the runbook major tracks the CLI major (self-mode) --------------
// A canonical entry is a fence-outside header blockquote `> `runbook-version: X.Y``;
// body-text or fenced occurrences are never matched. RUNBOOK.md is canonical;
// RUNBOOK_cn.md, when present, must independently carry the same major.
function stripFencesLocal(text) {
  return text.replace(/```[\s\S]*?```/g, '');
}
function checkRunbookVersion(root, files) {
  const fails = [];
  let pkgMajor;
  try { pkgMajor = String(JSON.parse(fs.readFileSync(path.join(root, 'package.json'), 'utf8')).version || '').split('.')[0]; }
  catch { return fails; }               // no readable package.json → not this check's business
  if (!pkgMajor) return fails;
  for (const [label, required] of [['RUNBOOK.md', true], ['RUNBOOK_cn.md', false]]) {
    const text = files[label];
    if (text === undefined) {           // RUNBOOK.md is canonical and REQUIRED under --self; _cn is optional
      if (required) fails.push(`CK-11 ${label}: the canonical runbook is missing — a version header is required`);
      continue;
    }
    // header region only: before the first h2 — a body-text blockquote never counts
    const h2 = text.search(/^##\s/m);
    const header = h2 >= 0 ? text.slice(0, h2) : text;
    const hits = [...stripFencesLocal(header).matchAll(/^>\s*`runbook-version:\s*([^`]*)`/gm)];
    if (hits.length === 0) fails.push(`CK-11 ${label}: no \`runbook-version\` blockquote entry (expected \`> \`runbook-version: ${pkgMajor}.x\`\`)`);
    else if (hits.length > 1) fails.push(`CK-11 ${label}: ${hits.length} runbook-version entries — exactly one required`);
    else {
      const m = /^(\d+)\.(\d+)$/.exec(hits[0][1].trim());
      if (!m) fails.push(`CK-11 ${label}: malformed runbook-version value '${hits[0][1].trim()}' (expected X.Y)`);
      else if (m[1] !== pkgMajor) fails.push(`CK-11 ${label}: runbook-version major ${m[1]} != package major ${pkgMajor}`);
    }
  }
  return fails;
}

const PACKAGED_RUNBOOK = path.join(__dirname, '..', 'RUNBOOK.md');
function checkRunbookFreshness(root, packagedPath = PACKAGED_RUNBOOK) {
  const scaffolded = path.join(root, 'apriori', 'runbook.md');
  if (!fs.existsSync(scaffolded) || !fs.existsSync(packagedPath)) return [];
  if (fs.readFileSync(scaffolded, 'utf8') === fs.readFileSync(packagedPath, 'utf8')) return [];
  return [`apriori/runbook.md differs from the installed apriori-cli runbook — run 'apriori update' to refresh`];
}

test('CK-11 the runbook major tracks the CLI major', () => {
  const { checkRunbookVersion } = require('../lib/check');
  const os3 = require('node:os'); const fs3 = require('node:fs'); const p3 = require('node:path');
  const root = fs3.mkdtempSync(p3.join(os3.tmpdir(), 'apriori-ck11-'));
  fs3.writeFileSync(p3.join(root, 'package.json'), JSON.stringify({ version: '4.0.3' }));
  const EN = '# R\n\n> `runbook-version: 4.0` · upstream: x\n';
  const CN = '# R\n\n> `runbook-version: 4.0` · 上游:x\n';
  const files = (en, cn) => ({ 'RUNBOOK.md': en, ...(cn !== undefined ? { 'RUNBOOK_cn.md': cn } : {}) });

  assert.deepStrictEqual(checkRunbookVersion(root, files(EN, CN)), []);

});
