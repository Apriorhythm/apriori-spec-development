   418	function closedFenceSpans(text) {

   427	// heading line never participates. Scenario DELIMITERS are found exactly the state-A way:
   428	// scan the fence-STRIPPED view (stripFences removes closed pairs, which can expose a heading
   429	// that shares its raw line with an inline closed fence), then map each heading back to its
   430	// raw position; body lines come from the RAW text between consecutive heading lines (fenced
   431	// content therefore participates in comparison, per the engine contract).
   432	function scanBlockStructure(text) {
   433	  const clean = text.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
   434	  const spans = closedFenceSpans(clean);
   435	  const stripped = clean.replace(/```[\s\S]*?```/g, '');
   436	  // stripped offset → raw offset (piecewise: add back the lengths of spans passed)
   437	  const toRaw = (sOff) => {
   438	    let removed = 0;
   439	    for (const [a, b] of spans) {
   440	      if (sOff + removed >= a) removed += b - a;
   441	      else break;
   442	    }
   443	    return sOff + removed;
   444	  };
   445	  const lines = clean.split('\n');
   446	  const lineStart = [];
   447	  { let off = 0; for (const l of lines) { lineStart.push(off); off += l.length + 1; } }
   448	  const rawLineOf = (rawOff) => {
   449	    let lo = 0, hi = lineStart.length - 1, ans = 0;
   450	    while (lo <= hi) { const mid = (lo + hi) >> 1; if (lineStart[mid] <= rawOff) { ans = mid; lo = mid + 1; } else hi = mid - 1; }
   451	    return ans;
   452	  };
   453	  // headings in the state-A view
   454	  const headRe = /^####[^\S\n]+Scenario:[^\S\n]+(.*)$/gm;
   455	  const heads = [];
   456	  let hm;
   457	  while ((hm = headRe.exec(stripped)) !== null)
   458	    heads.push({ title: hm[1], rawLine: rawLineOf(toRaw(hm.index)) });
   459	  const startLine = /^###\s+Requirement:/.test(lines[0] || '') ? 1 : 0;   // heading never compared
   460	  const collect = (from, to) => {
   461	    const items = [];
   462	    for (let i = from; i < to; i++) {
   463	      const n = norml(lines[i]);
   464	      if (n) items.push({ raw: lines[i], norm: n });
   465	    }
   466	    return items;
   467	  };
   468	  const firstHead = heads.length ? heads[0].rawLine : lines.length;
   469	  const prose = collect(startLine, firstHead);
   470	  const scenarios = heads.map((h, i) => ({
   471	    title: h.title,
   472	    body: collect(h.rawLine + 1, i + 1 < heads.length ? heads[i + 1].rawLine : lines.length),
   473	  }));
   474	  return { prose, scenarios };

   479	function missingFrom(oldItems, newItems) {
   480	  const missing = [];

   491	// → { retained, titleChanged, dropped, added, ambiguous, missingLines } (req B4 shapes/order)
   492	function compareModifiedBlock(oldBlock, newBlock, idOf) {
   493	  const empty = { retained: [], titleChanged: [], dropped: [], added: [], ambiguous: [], missingLines: [] };
   494	  if (oldBlock.trim() === newBlock.trim()) return { ...empty };   // idempotent fast path — no scan, no idOf
   495	  const o = scanBlockStructure(oldBlock);
   496	  const n = scanBlockStructure(newBlock);
   497	  const keyOf = (title) => { const id = idOf(title); return id !== null ? { key: id, id } : { key: norml(title), id: null }; };
   498	  const group = (list) => {
   499	    const m = new Map();
   500	    list.forEach((sc, idx) => {
   501	      const { key, id } = keyOf(sc.title);
   502	      if (!m.has(key)) m.set(key, { id, occ: [] });
   503	      m.get(key).occ.push({ ...sc, idx });
   504	    });
   505	    return m;
   506	  };
   507	  const om = group(o.scenarios), nm = group(n.scenarios);
   508	  const res = { retained: [], titleChanged: [], dropped: [], added: [], ambiguous: [], missingLines: [] };
   509	  // requirement prose first
   510	  for (const line of missingFrom(o.prose, n.prose)) res.missingLines.push({ scenario: null, line });
   511	  const ambON = [], ambNewOnly = [];
   512	  for (const [key, og] of om) {
   513	    const ng = nm.get(key);
   514	    const oc = og.occ.length, nc = ng ? ng.occ.length : 0;
   515	    if (oc === 1 && nc === 1) {
   516	      const oldSc = og.occ[0], newSc = ng.occ[0];
   517	      const entry = { id: og.id, title: newSc.title };
   518	      if (norml(oldSc.title) === norml(newSc.title)) res.retained.push(entry);
   519	      else res.titleChanged.push({ id: og.id, oldTitle: oldSc.title, newTitle: newSc.title });
   520	      const scenLabel = og.id !== null ? og.id : oldSc.title;
   521	      for (const line of missingFrom(oldSc.body, newSc.body)) res.missingLines.push({ scenario: scenLabel, line });
   522	    } else if (oc >= 1 && nc === 0) {
   523	      if (oc === 1) res.dropped.push({ id: og.id, title: og.occ[0].title });
   524	      else ambON.push({ key, side: 'old', oldCount: oc, newCount: 0, idx: og.occ[0].idx });
   525	    } else {
   526	      // any side > 1 with the other >= 1
   527	      const side = oc > 1 && nc > 1 ? 'both' : (oc > 1 ? 'old' : 'new');
   528	      ambON.push({ key, side, oldCount: oc, newCount: nc, idx: og.occ[0].idx });
   529	    }
   530	  }
   531	  for (const [key, ng] of nm) {
   532	    if (om.has(key)) continue;
   533	    const nc = ng.occ.length;
   534	    if (nc === 1) res.added.push({ id: ng.id, title: ng.occ[0].title });
   535	    else ambNewOnly.push({ key, side: 'new', oldCount: 0, newCount: nc, idx: ng.occ[0].idx });
   536	  }
   537	  ambON.sort((a, b) => a.idx - b.idx);
   538	  ambNewOnly.sort((a, b) => a.idx - b.idx);
   539	  res.ambiguous = [...ambON, ...ambNewOnly].map(({ key, side, oldCount, newCount }) => ({ key, side, oldCount, newCount }));
   540	  return res;
   541	}

   550	function formatIntegrityHuman(entries) {

# STEP5 implementation review v4 — modified-block-integrity

## 结论摘要

IMPL-5 的 archive inline-fence 子问题已正确修复：

- archive collector 现在扫描 fence-stripped view；
- inline-fence titles 会进入 injected matcher；
- AM-47 已证明同 ID 的 old/new title 被分类为 `titleChanged`。

但 IMPL-5 不能整体标记为 `verified`。engine 与 archive 目前共享的是收窄后的 `[^\S\n]+` regex，而 verify/state A 使用规范明确钉死的 `\s+` regex。跨换行 whitespace 的合法 state-A delimiter 仍被错误归为 prose。

因此本轮仍有 1 个 formal gap。

## 已验证的修补部分

`pushIntegritySection()` 不再按 raw line 与 fence span 起始位置过滤标题，而是先执行：

```js
.replace(/```[\s\S]*?```/g, '')
```

再扫描 headings。这与 IMPL-4 的 inline closed-fence 语义一致。

新增 AM-47 测试通过 `archiveChange({idMatcherFactory})` 验证：

```text
```x```#### Scenario: KV-01 old title
```

与：

```text
```x```#### Scenario: KV-01 new title
```

均进入 matcher，最终 human output 包含 ID-paired `titleChanged`，不再退化成 `dropped + added`。该子问题已关闭。

## Remaining formal gap

### IMPL-5 — shared delimiter regex 仍不等同于 state-A actual regex

Risk: med

SPEC-DOC 明确规定 scenario delimiter 沿用 state A 的实际 regex：

```js
/^####\s+Scenario:\s+(.*)$/m
```

verify 当前也确实使用：

```js
/^####\s+Scenario:\s+(.*)$/gm
```

但 engine 与 archive collector 共同使用：

```js
/^####[^\S\n]+Scenario:[^\S\n]+(.*)$/gm
```

`[^\S\n]+` 排除了 newline，`\s+` 则包含 newline。因此“engine、archive、verify one shared semantic”的声明尚未成立。

两个仍可复现的反例：

```text
### Requirement: R
####
Scenario: KV-01 split-a
- body
```

```text
### Requirement: R
#### Scenario:
KV-02 split-b
- body
```

State A/verify 分别识别：

```json
["KV-01 split-a"]
```

```json
["KV-02 split-b"]
```

replacement 删除这些内容时，正确 integrity classification 应包含对应 `dropped` scenario。当前 engine 实际返回：

```json
{
  "dropped": [],
  "missingLines": [
    {"scenario":null,"line":"…"}
  ]
}
```

即把 scenario loss 错归为 requirement prose loss。

318/318 GREEN 的原因是新增 AM-47 只覆盖 inline fence，没有覆盖 state-A `\s+` 跨换行行为。

## 建议修复与 oracle

应让 engine 与 archive collector 使用和 verify 完全相同的 shared regex，而不是三个文本相近的独立定义：

```js
/^####\s+Scenario:\s+(.*)$/gm
```

raw-boundary mapping 还需正确处理一个 delimiter 横跨多条 raw lines 的情形：delimiter 自身占用的所有 raw lines不得进入 prose 或 scenario body。

建议补充：

- `####\nScenario: KV-01 title`；
- `#### Scenario:\nKV-02 title`；
- 上述形式经过 archive injected matcher 后仍按 ID 配对；
- engine、archive collector、verify 对同一 corpus 得到完全相同的 title sequence。

## 其余 P8 复核

未发现其他新的 spec-vs-code gap：

- IMPL-1：raw `missingLines` 正确；
- IMPL-2：assembled human line 截断正确；
- IMPL-3：repaired rename-then-modify pair 正确；
- IMPL-4：fence-stripped delimiter discovery 与 raw body comparison 正确；
- MBI-1：retained=9、titleChanged=0、两条旧 THEN line 的修正正确；
- JSON presence/absence、two-batch contract、preflight insertion、sanitization 与 informative-only data flow 未出现回归。
