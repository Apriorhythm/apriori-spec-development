// The default matcher factory for a caller that did not inject one (the bin seam does). The
// controlled matcher lives in spec-runner, which requires THIS module at load time — so it is
// resolved lazily, at call time, never at load: the module graph stays acyclic. Nothing here
// compiles a config pattern in-process.
function defaultIdMatcherFactory(cwd) {
  const r = require('./config').resolveIdPattern(cwd, null);
  if (r.error) return { error: r.error };
  return require('./spec-runner').makeIdMatcher(r);
}
function matcherFor(factory, cwd) {
  let m;
  try { m = (factory || defaultIdMatcherFactory)(cwd); } catch (e) { return { error: String(e.message || e) }; }
  if (!m) return { error: 'no id matcher' };
  return m;
}


// ((cwd) => matcher | {error}) so this module never requires spec-runner; a missing factory
// degrades exactly like a factory error — one bounded stderr warning, report skipped,
// every other archive output/exit/write unchanged.
// → { entries } | { warning } — the entries of every MODIFIED pair, or the one reason none could be built
function integrityEntries(modifiedBlocks, idMatcherFactory, cwd) {
  if (!modifiedBlocks || modifiedBlocks.size === 0) return { entries: [] };
  if (!idMatcherFactory) return { warning: 'id matcher unavailable (no factory injected)' };
  let matcher;
  try { matcher = idMatcherFactory(cwd); } catch (e) { return { warning: 'id-pattern resolution failed: ' + String(e.message || e) }; }
  if (!matcher || matcher.error) return { warning: 'id-pattern resolution failed: ' + String(matcher && matcher.error) };
  // one batch over every old+new block scenario title — discovered the SAME state-A way
  // the engine does (fence-STRIPPED view: an inline closed fence can expose a heading)
  const titles = [];
  const collect = (text) => {
    const stripped = text.replace(/\r\n/g, '\n').replace(/\r/g, '\n').replace(/```[\s\S]*?```/g, '');
    let m;
    const re = /^####\s+Scenario:\s+(.*)$/gm;                 // the state-A regex, verbatim
    while ((m = re.exec(stripped)) !== null) titles.push(m[1]);
  };
  for (const list of modifiedBlocks.values()) for (const { oldBlock, newBlock } of list) { collect(oldBlock); collect(newBlock); }
  const res = matcher.batch(titles);
  if (res.failure) return { warning: 'matcher failure: ' + res.failure };
  const titleId = new Map();
  titles.forEach((t, i) => { if (!titleId.has(t)) titleId.set(t, res.ids[i]); });
  const idOf = (t) => { const v = titleId.get(t); return v === undefined ? null : v; };
  const entries = [];
  for (const [file, list] of modifiedBlocks)
    for (const { name, oldBlock, newBlock } of list)
      entries.push({ file, name, ...compareModifiedBlock(oldBlock, newBlock, idOf) });
  entries.sort((a, b) => (a.file === b.file ? (a.name < b.name ? -1 : 1) : (a.file < b.file ? -1 : 1)));
  return { entries };
}
