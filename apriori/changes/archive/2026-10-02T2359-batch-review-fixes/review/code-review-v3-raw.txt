const NOFOLLOW = fs.constants.O_NOFOLLOW || 0;
// O_NOFOLLOW guards the last component only: a parent directory swapped for a symlink after the
// obstruction check would still lead the open outside. So containment is judged on the opened
// descriptor itself — where it really is (/proc/self/fd on Linux), else its path's realpath right
// after the open (a narrower window, the residual on platforms without /proc).
function openedInside(root, abs, fd) {
  let real = null;
  try { real = fs.readlinkSync(`/proc/self/fd/${fd}`); } catch { /* no /proc */ }
  if (real === null) { try { real = fs.realpathSync(abs); } catch { return false; } }
  let realRoot;
  try { realRoot = fs.realpathSync(root); } catch { return false; }
  return real !== realRoot && (real + path.sep).startsWith(realRoot + path.sep);
}

function readPlain(root, abs) {
  let fd;
  try { fd = fs.openSync(abs, fs.constants.O_RDONLY | NOFOLLOW); }
  catch (e) { if (['ELOOP', 'ENOENT', 'EISDIR', 'ENOTDIR'].includes(e.code)) return null; throw e; }
  try {
    const st = fs.fstatSync(fd);
    return st.isFile() && openedInside(root, abs, fd) ? { bytes: fs.readFileSync(fd), st } : null;
  } finally { fs.closeSync(fd); }
}

// Rewrite the file readPlain saw — only while it is still that file: opened without following a
// symlink at the last component, inside the project (judged on the descriptor), the same device and
// inode, and only then truncated. false = it was replaced or redirected in between (nothing written).
function writePlain(root, abs, buf, seen) {
  let fd;
  try { fd = fs.openSync(abs, fs.constants.O_WRONLY | NOFOLLOW); }
  catch (e) { if (['ELOOP', 'ENOENT', 'EISDIR', 'ENOTDIR'].includes(e.code)) return false; throw e; }
  try {
    const st = fs.fstatSync(fd);
    if (!st.isFile() || st.dev !== seen.st.dev || st.ino !== seen.st.ino || !openedInside(root, abs, fd)) return false;
    fs.ftruncateSync(fd, 0);
    for (let off = 0; off < buf.length;) off += fs.writeSync(fd, buf, off, buf.length - off, off);
    return true;
  } finally { fs.closeSync(fd); }
}

1. With descriptor lookup unavailable, redirect `apriori/guides` outside before the read open. The external file contains a recognized earlier guide edition.
2. Restore the original directory before the fallback `realpathSync(abs)`. Containment passes for the restored internal pathname, while `readFileSync(fd)` reads the external file.
3. Repeat the redirect-and-restore around the write open and its fallback containment check.
4. The device/inode comparison passes: both descriptors refer to the same external file. `ftruncateSync(fd, 0)` then truncates it and the subsequent write overwrites it.

This retains BRF-R4’s substantive risk: adoption can overwrite an existing external file. Moving the pathname check after opening narrows the window but does not bind containment to the descriptor.

The fallback also runs on **any** `/proc/self/fd` readlink failure, including Linux environments where that lookup is unavailable; it is not confined to non-Linux platforms.

**Contract and evidence.** The updated requirement still promises that the opened file is inside the project and that update never writes through a path resolving outside it.
