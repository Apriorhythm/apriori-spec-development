test('TX-01 the shared fence grammar: tildes by line, marker and length; backticks as always', () => {
  const S = text.stripFences;
  // a tilde fence hides what it holds, structure-shaped lines included
  assert.strictEqual(S('a\n~~~\n#### Scenario: XX-99 fake\n~~~\nb'), 'a\n\nb');
  // an info string may follow the opener; the closer may be longer, indented up to three spaces, trailing blanks allowed
  assert.strictEqual(S('~~~ md\nX\n   ~~~~  \nY'), '\nY');
  // a shorter run, a tagged line or the other marker does not close it
  assert.strictEqual(S('~~~~\n~~~\nX\n~~~ md\n```\n~~~~\nY'), '\nY');
  // inline ~~~ is ordinary text, and so is a line indented four spaces
  assert.strictEqual(S('say `~~~` twice: `~~~`'), 'say `~~~` twice: `~~~`');
  assert.strictEqual(S('    ~~~\nX\n    ~~~'), '    ~~~\nX\n    ~~~');
  // an unclosed tilde opener is ordinary text, as an unclosed backtick one is
  assert.strictEqual(S('~~~\nX'), '~~~\nX');
  assert.strictEqual(S('```\nX'), '```\nX');
  // whichever opens first wins; inside one, the other marker is content
  assert.strictEqual(S('```\n~~~\n```\nY\n~~~'), '\nY\n~~~');
  assert.strictEqual(S('~~~\n```\n~~~\nY\n```'), '\nY\n```');
  // CRLF lines read the same
  assert.strictEqual(S('~~~\r\nX\r\n~~~\r\nY'), '\nY');
  // spans are what stripFences removes, in order
  const t = 'a\n```\nx\n```\nb\n~~~\ny\n~~~\nc';
  assert.deepStrictEqual(text.fenceSpans(t).map(([s, e]) => t.slice(s, e)), ['```\nx\n```', '~~~\ny\n~~~']);
  // the line readers' open/close rule: ``` as before (any indentation; any ```-led line closes), ~~~ by marker and length
  assert.deepStrictEqual(text.fenceOpen('  ```js'), { ch: '`', len: 3 });
  assert.deepStrictEqual(text.fenceOpen('~~~~ md'), { ch: '~', len: 4 });
  assert.strictEqual(text.fenceOpen('    ~~~'), null);
  assert.strictEqual(text.fenceCloses({ ch: '`', len: 3 }, '```js'), true);
  assert.strictEqual(text.fenceCloses({ ch: '~', len: 4 }, '~~~'), false);
  assert.strictEqual(text.fenceCloses({ ch: '~', len: 3 }, '~~~ md'), false);
  assert.strictEqual(text.fenceCloses({ ch: '~', len: 3 }, ' ~~~~ '), true);
  assert.strictEqual(text.fenceCloses({ ch: '~', len: 3 }, '```'), false);
});

# Delta — text (fence-tilde-readers)

## ADDED Requirements

### Requirement: one fence grammar for every structural reader
`lib/text.js` SHALL define the one fence grammar every structural reader of Markdown uses — this tool's own grammar, not CommonMark (lists, blockquotes and other containers are not modeled). Backticks keep the historical reading byte for byte: a run of three backticks anywhere opens and the next run of three closes, and an unclosed opener is ordinary text. Tildes open only as a line whose first non-blank characters (up to three spaces in) are three or more `~`, an info string allowed after them, and close at the first later line holding only `~` at least as many (up to three spaces in, trailing blanks allowed); an inline `~~~` is ordinary text. Whichever opens first wins, and inside a fence the other marker is content; CRLF lines read as LF lines. Span readers (`stripFences`, `fenceSpans`) treat an unclosed tilde opener as ordinary text, as they do a backtick one; line readers use the same open and close rule (`fenceOpen`, `fenceCloses`) and keep their own policy for an unterminated fence. The readers SHALL be: scenario discovery in `check` and in `verify` (one scanner, so they agree), the delta parser in `archive`, requirement-block boundaries in a store wherever they are read — the merge, the render, the MODIFIED baselines, the single-file form and the change-scope scan — walked exactly as the historical requirement regex walked them (a heading's name may run onto its next line; a block ends at the first later line-start `### Requirement:`), except that a heading inside a fence of either marker is block content, the block keeping its raw text; and the render keeps a store's text before its first live requirement — all of it when the store has none yet — and must read back as exactly its merged blocks, in order: when a part's text would swallow another (an unclosed fence opener in a kept preamble or an earlier block pairing with a fence further on), `archive` refuses that module in both forms, naming the first block that went missing, and writes nothing — the MODIFIED-block comparison engine, the process-config reader, gate's truth-doc `source-commit` reader (C6), and the review reader — verdict lines of a self-contained review document, and the round summary's opening ids and re-review conclusions. Existing Markdown that uses only backtick fences SHALL strip exactly as before, and every spec store and delta in this repository SHALL split into the same requirement blocks as before. The scan SHALL stay near-linear on any input: no position is searched again for a backtick run once passed, and an unclosed opener never rescans or refilters the rest of the text. Outside this grammar, by design: the state-file reader (`lib/flow.js`, which already reads tildes) and the golden-path script's Quickstart shell-block extraction (`scripts/golden-path.mjs`, which reads one known README).

// The rendered store must read back as exactly the merged blocks, in order. Rendering composes text —
// the store's preamble, then the blocks — and composition must never let one part's syntax swallow
// another: an unclosed fence opener in a preamble (or in a block) pairs with the first matching fence
// line after it, wherever that lands. -> null, or the refusal naming the first block that went missing
function renderReadback(rendered, store) {
  const want = [...store.values()];
  const got = [...parseRequirementsStrict(rendered).map.values()];
  for (let i = 0; i < want.length; i++) {
    if (got[i] === want[i]) continue;
    const name = (/^###\s+Requirement:\s+(.+?)\s*$/m.exec(want[i]) || [])[1] || `block ${i + 1}`;
    return `the merged store would not read back as its requirement blocks — '${name.trim()}' is swallowed or altered by the text before it (an unclosed fence opener in the store's preamble or an earlier block pairs with a fence line further on); close that fence in the store and rerun — nothing is written`;
  }
  return got.length === want.length ? null : `the merged store would not read back as its requirement blocks — ${got.length} read back for ${want.length} merged; nothing is written`;
}

function renderStore(storeText, store) {
  const first = requirementHeadings(storeText)[0];          // a fenced example heading is not where the store's blocks begin
  // the text before the first live requirement is kept as the store's preamble; a store with no live
  // requirement at all — empty, prose only, or only a fenced example — is ALL preamble, never dropped
  const header = first ? storeText.slice(0, first.at)
    : storeText.trim() ? storeText.replace(/\s*$/, '\n\n') : '';
  return header + [...store.values()].join('\n');
}

// Realpath containment of a FUTURE write target: a not-yet-existing target is judged by its
// nearest existing ancestor, and the root itself is NOT inside (strict). Fail-closed. NOT
// resolve's containsExistingPath (existing paths only, root counts as inside).
function containsFuturePath(root, target) {
  let realRoot;
  try { realRoot = fs.realpathSync(root); } catch { return false; }
  let cur = path.resolve(target);
  const rest = [];
  while (!fs.existsSync(cur)) {
    const parent = path.dirname(cur);
    if (parent === cur) return false;
    rest.unshift(path.basename(cur));
    cur = parent;
  }
  let realCur;
  try { realCur = fs.realpathSync(cur); } catch { return false; }
  const full = rest.length ? path.join(realCur, ...rest) : realCur;
  return full !== realRoot && (full + path.sep).startsWith(realRoot + path.sep);
}

    } else {
      console.log(`\nRESULT: MERGED (dry-run; ${store.size} requirements in result) — pass --write to apply`);
    }
    return 0;
  });
}

// High-level archive: the whole change, four phases (preflight → stage → commit → move).

function archiveChange(o) {
  const out = [], err = [];
  const ops = o.ops || { writeFileSync: fs.writeFileSync.bind(fs), renameSync: fs.renameSync.bind(fs), rmSync: fs.rmSync.bind(fs) };
  const cwd = o.cwd || process.cwd();

    } else {
      for (const sfx of p.unstampedMutations) out.push(`warning: ${unstampedWarning(sfx)}`);
      out.push(`note: cas waived by ${w.waiver}`);
    }
  }
  for (const n of p.notes) out.push(`note: ${n}`);

  // change adds or modifies must carry one stable id that nothing else carries — refused here,

      else {

        if (sp.debt.length) out.push(`note: store debt outside this change (reported, not a block): ${sp.debt.join('; ')}`);
      }

    for (const f of preflightFailures) err.push(`archive: ${f}`);

    return { code: 0, out, err };
  }

  // --- phase 1b: the content manifest, INSIDE the transaction and BEFORE the first store byte.
  // What the archive contained at archive time is recorded in the bundle itself, so it travels
  // with the move; `apriori check` reads it back to report drift. Written here — not after the
  // move — so a failure can never leave the stores rewritten and the archive unbaselined: if the
  // manifest cannot be produced (a file vanished between listing and hashing, an unreadable
  // entry, a write failure) the run stops with nothing written. The stamp is fixed now and the

  // failed move recomputes and replaces it (the manifest never lists itself).
  // The bundle root itself must be a real directory: written through a symlinked
  // `changes/<name>` the manifest would land in — and rewrite the baseline of — whatever the
  // link points at (an already archived bundle included), and the move would move the link.
  let rootStat = null;
  try { rootStat = fs.lstatSync(bundleDir); } catch (e) { rootStat = null; }
  if (!rootStat || rootStat.isSymbolicLink() || !rootStat.isDirectory()) {
    err.push(`archive: ${bundleDir} is ${rootStat ? (rootStat.isSymbolicLink() ? 'a symbolic link' : 'not a directory') : 'not readable'} — an in-flight bundle must be a real directory; nothing written`);
    out.push('\nRESULT: REFUSED — nothing written');
    return { code: 1, out, err };
  }

  let manifestEntries = 0;
  try {
    const tmpName = manifest.ownTempName();
    const m = manifest.build(bundleDir, { change: o.change, stamp: archiveStamp(now), ops, exclude: [tmpName] });
    manifestEntries = Object.keys(m.files).length;

  } catch (e) {

    out.push('\nRESULT: FAILED — nothing written');
    return { code: 1, out, err };
  }
  out.push(`manifest: ${manifest.MANIFEST_NAME} (${manifestEntries} entries)`);

  // --- phase 2: stage (this run's temps only — preflight proved none pre-existed) ---
  const staged = [];
  try {
    for (const j of jobs) {
      fs.mkdirSync(path.dirname(j.store), { recursive: true });
      ops.writeFileSync(j.tmp, j.text);
      staged.push(j.tmp);
    }
  } catch (e) {
    for (const t of staged) { try { ops.rmSync(t, { force: true }); } catch { /* best effort */ } }
    err.push(`archive: staging failed (${e.message}) — stores untouched, this run's temp files removed`);
    return { code: 1, out, err };
  }

  // --- phase 3: commit (per-file atomic renames, sorted order; no rollback on mid-commit failure) ---
  const committed = [];
  for (const j of jobs) {
    try { ops.renameSync(j.tmp, j.store); committed.push(j); }
    catch (e) {
      err.push(`archive: COMMIT FAILED at ${j.suffix} (${e.message})`);
      err.push(`  committed:     ${committed.map((c) => c.suffix).join(', ') || '(none)'}`);
      const remaining = jobs.filter((x) => !committed.includes(x));
      err.push(`  NOT committed: ${remaining.map((x) => x.suffix).join(', ')}`);
      err.push(`  temp files remaining for manual completion: ${remaining.map((x) => x.tmp).join(', ')}`);
      return { code: 1, out, err };
    }
  }

  // --- phase 4: move. The DEFAULT since batch C row 8: the runbook always declared that
  // archive "merges the delta specs into apriori/specs and moves the bundle", but the move
  // used to happen only under an explicit --changes-dir (P6). The behaviour is aligned to

  let moved = null;

  catch (e) { err.push(`archive: stores committed but the change-dir move failed: ${e.message} — rerun to complete`); return { code: 1, out, err }; }

  out.push(`\nRESULT: MERGED — ${jobs.length} module store(s) rewritten${moved ? `; change archived → ${moved}` : ''}`);
  return { code: 0, out, err };
}

// Pure merge. Returns { store: Map, merged, modified, deprecated, renamed, conflicts, unchanged }
function merge(storeText, delta, change) {
  const parsed = parseRequirementsStrict(storeText);
  const store = parsed.map;
  const merged = [], modified = [], deprecated = [], renamed = [], conflicts = [], unchanged = [];
  // a corrupt store (same requirement name twice) must never be silently collapsed
  for (const d of parsed.duplicates) conflicts.push(`duplicate requirement '${d}' in the store (corrupt store)`);
  const renamedTargets = new Set();   // names created by THIS run's renames — an ADDED hitting one is a same-delta collision, never a rerun no-op
  // RENAMED first: rename the store block's ID, preserving its content
  for (const [oldName, newName] of (delta.RENAMED || [])) {
    if (!store.has(oldName)) {
      // rerun signature: source already gone AND target present → this rename already happened
      if (store.has(newName)) { unchanged.push(`${oldName} -> ${newName} (already renamed)`); continue; }
      conflicts.push(`RENAMED '${oldName}' has no target in store`); continue;
    }
    if (store.has(newName)) { conflicts.push(`RENAMED target '${newName}' already exists in store`); continue; }
    const block = store.get(oldName);
    const renamedBlock = block.replace(/^###\s+Requirement:\s+.+$/m, `### Requirement: ${newName}`);
    // rebuild the map preserving order, swapping the key at the old position
    const rebuilt = new Map();
    for (const [k, v] of store) rebuilt.set(k === oldName ? newName : k, k === oldName ? renamedBlock : v);
    store.clear();
    for (const [k, v] of rebuilt) store.set(k, v);
    renamed.push(`${oldName} -> ${newName}`);
    renamedTargets.add(newName);
  }
  for (const [name, block] of delta.ADDED) {
    if (store.has(name)) {
      // idempotent rerun: identical content means this delta already merged — a no-op, not a collision.
      // But a name just created by THIS run's RENAMED is a same-delta collision, never a rerun.
      if (!renamedTargets.has(name) && store.get(name).trim() === block.trim()) unchanged.push(name);
      else conflicts.push(`ADDED '${name}' already exists in store`);
    } else { store.set(name, block); merged.push(name); }
  }
  for (const [name, block] of delta.MODIFIED) {
    if (!store.has(name)) conflicts.push(`MODIFIED '${name}' has no target in store`);
    // already applied (rerun signature) — same trim comparison the ADDED no-op uses
    else if (store.get(name).trim() === block.trim()) unchanged.push(name);
    else { store.set(name, block); modified.push(name); }
  }
  for (const name of delta.REMOVED.keys()) {
    if (!store.has(name)) {
      // rerun signature: already deprecated BY THIS CHANGE → no-op; by another change → real collision
      const depByThis = `${name}  _deprecated (superseded by ${change})_`;
      if (store.has(depByThis)) { unchanged.push(`${name} (already deprecated)`); continue; }
      const depOther = [...store.keys()].find((k) => k.startsWith(`${name}  _deprecated (superseded by `));
      if (depOther) { conflicts.push(`REMOVED '${name}' was already deprecated by a different change ('${depOther}')`); continue; }
      conflicts.push(`REMOVED '${name}' has no target in store`);
    } else {
      const block = store.get(name);
      const nl = block.indexOf('\n');
      store.set(name, block.slice(0, nl) + `  _deprecated (superseded by ${change})_` + block.slice(nl));
      deprecated.push(name);
    }
  }
  return { store, merged, modified, deprecated, renamed, conflicts, unchanged };
}
